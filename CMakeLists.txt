  message(FATAL_ERROR "Prepare pinned dependencies and set ESL_DEPS_DIR")
endif()
function(check_dependency name expected)
  if(NOT EXISTS "${ESL_DEPS_DIR}/${name}/.esl-dependency")
    message(FATAL_ERROR "Missing verification marker for ${name}; run prepare_dependencies.py")
  endif()
  file(READ "${ESL_DEPS_DIR}/${name}/.esl-dependency" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Wrong dependency version/hash for ${name}")
  endif()
endfunction()

check_dependency(tomlplusplus "3.4.0 8517f65938a4faae9ccf8ebb36631a38c1cadfb5efa85d9a72e15b9e97d25155")
