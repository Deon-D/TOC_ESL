# Also usable as a dependency-free, standalone RN/HN/SN build.
cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(chi_vip LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(ESL_SANITIZERS "Enable address/undefined sanitizers" OFF)
  add_library(esl_chi ../chi_interface/src/chi_port.cpp)
  target_include_directories(esl_chi PUBLIC ../chi_interface/include)
  target_compile_options(esl_chi PRIVATE -Wall -Wextra -Wpedantic -Werror)
  enable_testing()
endif()
add_library(esl_chi_vip src/types.cpp src/rules.cpp rn/src/rn.cpp rn/src/request_engine.cpp rn/src/snoop_engine.cpp
  hn/src/hn.cpp sn/src/sn.cpp sn/src/memory_backend.cpp)
target_include_directories(esl_chi_vip PUBLIC include rn/include hn/include sn/include ../chi_interface/include)
target_compile_options(esl_chi_vip PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(esl_chi_vip_transport src/transport_adapter.cpp src/node_transport_adapter.cpp)
target_link_libraries(esl_chi_vip_transport PUBLIC esl_chi_vip esl_chi)
target_compile_options(esl_chi_vip_transport PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(TARGET esl)
  add_library(esl_chi_vip_adapter src/esl_adapter.cpp src/system_esl_adapter.cpp)
  target_link_libraries(esl_chi_vip_adapter PUBLIC esl_chi_vip_transport esl)
  target_compile_options(esl_chi_vip_adapter PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_library(esl_chi_vip_system system/src/system.cpp)
  target_include_directories(esl_chi_vip_system PUBLIC system/include)
  target_link_libraries(esl_chi_vip_system PUBLIC esl_chi_vip_adapter)
  target_compile_options(esl_chi_vip_system PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(esl_chi_vip_runner_tests ../../tests/chi_vip_runner_test.cpp)
  target_link_libraries(esl_chi_vip_runner_tests PRIVATE esl_chi_vip_system)
  target_compile_options(esl_chi_vip_runner_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  foreach(case runner_read runner_mixed runner_determinism runner_invalid)
    add_test(NAME chi_vip_${case} COMMAND esl_chi_vip_runner_tests ${case})
    set_tests_properties(chi_vip_${case} PROPERTIES TIMEOUT 60)
  endforeach()
  add_executable(esl_chi_vip_platform_tests ../../tests/chi_vip_platform_test.cpp)
  target_link_libraries(esl_chi_vip_platform_tests PRIVATE esl_chi_vip_adapter)
  target_compile_options(esl_chi_vip_platform_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME chi_vip_platform COMMAND esl_chi_vip_platform_tests)
  set_tests_properties(chi_vip_platform PROPERTIES TIMEOUT 60)
  add_executable(esl_chi_vip_system_platform_tests ../../tests/chi_vip_system_platform_test.cpp)
  target_link_libraries(esl_chi_vip_system_platform_tests PRIVATE esl_chi_vip_adapter)
  target_compile_options(esl_chi_vip_system_platform_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME chi_vip_system_platform COMMAND esl_chi_vip_system_platform_tests)
  set_tests_properties(chi_vip_system_platform PROPERTIES TIMEOUT 60)
endif()
if(ESL_SANITIZERS)
  if(NOT TARGET esl)
    target_compile_options(esl_chi PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(esl_chi PUBLIC -fsanitize=address,undefined)
  endif()
  foreach(lib esl_chi_vip esl_chi_vip_transport)
    target_compile_options(${lib} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${lib} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
add_executable(esl_chi_vip_tests ../../tests/chi_vip_test.cpp ../../tests/chi_vip_contract_test.cpp)
target_link_libraries(esl_chi_vip_tests PRIVATE esl_chi_vip_transport)
target_compile_options(esl_chi_vip_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(case requests snoops hazards retry resources negative stress adapter contract)
  add_test(NAME chi_vip_${case} COMMAND esl_chi_vip_tests ${case})
  set_tests_properties(chi_vip_${case} PROPERTIES TIMEOUT 60)
endforeach()
add_executable(esl_chi_vip_sn_tests ../../tests/chi_vip_sn_test.cpp)
target_link_libraries(esl_chi_vip_sn_tests PRIVATE esl_chi_vip)
target_compile_options(esl_chi_vip_sn_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME chi_vip_sn COMMAND esl_chi_vip_sn_tests)
set_tests_properties(chi_vip_sn PROPERTIES TIMEOUT 60)
add_executable(esl_chi_vip_hn_tests ../../tests/chi_vip_hn_test.cpp)
target_link_libraries(esl_chi_vip_hn_tests PRIVATE esl_chi_vip)
target_compile_options(esl_chi_vip_hn_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME chi_vip_hn COMMAND esl_chi_vip_hn_tests)
set_tests_properties(chi_vip_hn PROPERTIES TIMEOUT 60)
add_executable(esl_chi_vip_system_tests ../../tests/chi_vip_system_test.cpp)
target_link_libraries(esl_chi_vip_system_tests PRIVATE esl_chi_vip_transport)
target_compile_options(esl_chi_vip_system_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(case basic coherence opcodes flow stress error)
  add_test(NAME chi_vip_system_${case} COMMAND esl_chi_vip_system_tests ${case})
  set_tests_properties(chi_vip_system_${case} PROPERTIES TIMEOUT 60)
endforeach()
